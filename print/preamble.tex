% mdbook-pandoc can emit a source chapter as either a chapter or a section,
% depending on its SUMMARY nesting and leading metadata. Give both commands the
% same B5-aware appearance; the Lua filter supplies the recto page break.
\titleformat{\chapter}[block]
  {\centering\LARGE\bfseries}
  {}{0pt}{}
\titlespacing*{\chapter}{0pt}{25mm}{18mm}
\titleformat{\section}[block]
  {\centering\LARGE\bfseries}
  {}{0pt}{}
\titlespacing*{\section}{0pt}{25mm}{18mm}

% Figure captions in the Markdown already begin with their original labels,
% such as "图 3.1". Do not add a second automatic figure label.
\captionsetup[figure]{
  labelformat=empty,
  font=small,
  justification=centering
}
\captionsetup[table]{
  labelformat=empty,
  font=small
}

% Keep dense component tables readable on the B5 page.
\AtBeginEnvironment{longtable}{%
  \small
  \setlength{\tabcolsep}{3.5pt}%
  \renewcommand{\arraystretch}{1.12}%
}
